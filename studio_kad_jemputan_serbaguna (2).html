⚙️ Kod Google Apps Script (1 Sheet Untuk Semua Majlis)

Buka Google Sheets baru > pergi ke Extensions > Apps Script. Padam semua kod sedia ada, gantikan dengan kod di bawah, kemudian klik Deploy > New deployment > Web app (Execute as: Me, Who has access: Anyone). Salin URL yang diberikan ke dalam tab Hubungi / RSVP. 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Format Lajur: Timestamp | Kod Majlis | Nama | Status Kehadiran | Bilangan Pax | Ucapan & Doa
  sheet.appendRow([
    new Date(),
    data.kodMajlis || "DEFAULT",
    data.nama || "",
    data.hadir || "",
    data.bilangan || "1",
    data.pesanan || ""
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var kod = e.parameter.kod || "";
  var out = [];
  
  // Baris pertama (index 0) adalah tajuk lajur
  for (var i = 1; i < rows.length; i++) {
    if (!kod || String(rows[i][1]).toUpperCase() === kod.toUpperCase()) {
      out.push({
        timestamp: rows[i][0],
        kodMajlis: rows[i][1],
        nama: rows[i][2],
        hadir: rows[i][3],
        bilangan: rows[i][4],
        pesanan: rows[i][5]
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}
Lajur dijana secara automatik apabila rekod pertama masuk.
